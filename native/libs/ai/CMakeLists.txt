if(THCAD_WITH_LLAMA)
  find_package(llama CONFIG QUIET)
  if(llama_FOUND)
    message(STATUS "llama.cpp ${LLAMA_VERSION}: local AI backend enabled")
    target_sources(thcad_ai PRIVATE src/llama_runtime.cpp)
    target_link_libraries(thcad_ai PRIVATE llama)
    target_compile_definitions(thcad_ai PUBLIC THCAD_HAS_LLAMA=1)
  else()
    message(WARNING "llama.cpp not found — local AI backend disabled")
  endif()
endif()
