# Licence keys (Ed25519 via libsodium): verification + machine code. Used by the app and the keygen.
# libsodium: vcpkg (Windows) provides unofficial-sodium; Homebrew (macOS) — the static library is preferred so
# the app bundle carries no extra dylib.
find_package(unofficial-sodium CONFIG QUIET)
if(TARGET unofficial-sodium::sodium)
  set(THCAD_SODIUM_TARGET unofficial-sodium::sodium)
else()
  find_path(SODIUM_INCLUDE_DIR sodium.h HINTS /opt/homebrew/opt/libsodium/include /usr/local/opt/libsodium/include)
  find_library(SODIUM_LIBRARY NAMES libsodium.a sodium libsodium HINTS /opt/homebrew/opt/libsodium/lib /usr/local/opt/libsodium/lib)
  if(SODIUM_INCLUDE_DIR AND SODIUM_LIBRARY)
    add_library(thcad_sodium INTERFACE)
    target_include_directories(thcad_sodium INTERFACE ${SODIUM_INCLUDE_DIR})
    target_link_libraries(thcad_sodium INTERFACE ${SODIUM_LIBRARY})
    if(SODIUM_LIBRARY MATCHES "\\.a$" AND NOT WIN32)
      find_package(Threads REQUIRED)
      target_link_libraries(thcad_sodium INTERFACE Threads::Threads)
    endif()
    set(THCAD_SODIUM_TARGET thcad_sodium)
  endif()
endif()

if(NOT THCAD_SODIUM_TARGET)
  if(THCAD_LICENSE_ENFORCE)
    message(FATAL_ERROR "libsodium not found: required for THCAD_LICENSE_ENFORCE=ON (brew install libsodium / vcpkg libsodium)")
  endif()
  message(WARNING "libsodium not found — licence library disabled")
  set(THCAD_HAS_LICENSE OFF PARENT_SCOPE)
  return()
endif()
message(STATUS "libsodium: ${THCAD_SODIUM_TARGET} ${SODIUM_LIBRARY}")

add_library(thcad_license STATIC src/license.cpp src/machine_id.cpp)
add_library(thcad::license ALIAS thcad_license)
target_include_directories(thcad_license PUBLIC include)
target_link_libraries(thcad_license PUBLIC ${THCAD_SODIUM_TARGET})
if(APPLE)
  target_link_libraries(thcad_license PRIVATE "-framework IOKit" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(thcad_license PRIVATE advapi32)
endif()

# signing: the owner's keygen and the unit tests only — the app never links this
add_library(thcad_license_issue STATIC src/issue.cpp)
add_library(thcad::license_issue ALIAS thcad_license_issue)
target_link_libraries(thcad_license_issue PUBLIC thcad_license)
set(THCAD_HAS_LICENSE ON PARENT_SCOPE)
